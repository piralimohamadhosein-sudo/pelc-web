   157→   157→/** اعمال سند تازه از تاریخچه + حفظ انتخاب‌های معتبر */
   158→   158→function syncFromHistory(
   159→   159→  set: (partial: Partial<EditorState>) => void,
   160→   160→  get: () => EditorState,
   161→   161→  doc: PelcDocument,
   162→   162→) {
   163→   163→  const h = history!
   164→   164→  const sel = get().selectedLayerId
   165→   165→  const layer = doc.layers.some((l) => l.id === sel) ? sel : (doc.layers[0]?.id ?? null)
   166→   166→  // حفظ انتخاب چندگانهٔ معتبر — شناسه‌های حذف‌شده کنار می‌روند
   167→   167→  const multi = get().selectedLayerIds.filter((id) => doc.layers.some((l) => l.id === id))
   168→   168→  const keepStart = get().selectedClusterStart
   169→   169→  let nextStart = keepStart
   170→   170→  if (keepStart !== null) {
   171→   171→    const l = doc.layers.find((x) => x.id === layer)
   172→   172→    const text = l?.text?.text ?? ""
   173→   173→    if (!l?.text || !letterClusters(text).some((c) => c.start === keepStart)) nextStart = null
   174→   174→  }
   175→   175→  set({
   176→   176→    doc,
   177→   177→    historyVersion: h.version,
   178→   178→    stepCount: h.stepCount,
   179→   179→    position: h.position,
   180→   180→    canUndo: h.canUndo,
   181→   181→    canRedo: h.canRedo,
   182→   182→    labels: h.labels(),
   183→   183→    selectedLayerId: layer,
   184→   184→    selectedLayerIds: multi,
   185→   185→    selectedClusterStart: nextStart,
   186→   186→  })
   187→   187→}

   217→   217→export const useEditorStore = create<EditorState>((set, get) => {
   218→   218→  /** اعمال جهش متن با بازنگاشت perLetter و پیگیری انتخاب حرف */
   219→   219→  const commitTextMutation = (mutation: TextMutation, label: string, extraPerLetter?: Record<string, LetterOverride>) => {
   220→   220→    const state = get()
   221→   221→    const layer = findLayer(state.doc, state.selectedLayerId)
   222→   222→    if (!layer?.text) return
   223→   223→    const base = extraPerLetter ?? layer.text.perLetter
   224→   224→    const perLetter = remapPerLetter(base, mutation)
   225→   225→    const next = withLayer(state.doc, layer.id, (l) => ({
   226→   226→      ...l,
   227→   227→      text: { ...l.text!, text: mutation.newText, perLetter },
   228→   228→    }))
   229→   229→    const keepStart = state.selectedClusterStart
   230→   230→    state.commit(next, label)
   231→   231→    if (keepStart !== null) {
   232→   232→      const mapped = keepStart >= mutation.shiftFrom ? keepStart + mutation.delta : keepStart
   233→   233→      const clusters = letterClusters(mutation.newText)
   234→   234→      if (clusters.some((c) => c.start === mapped)) {
   235→   235→        set({ selectedClusterStart: mapped })
   236→   236→      }
   237→   237→    }
   238→   238→  }
   239→   239→
   240→   240→  const initialDoc = newDocument()
   241→   241→  history = new EditorHistory(initialDoc, DEFAULT_CAPACITY)
   242→   242→
   243→   243→  return {
   244→   244→    doc: initialDoc,
   245→   245→    historyVersion: 0,
   246→   246→    stepCount: 0,
   247→   247→    position: 0,
   248→   248→    canUndo: false,
   249→   249→    canRedo: false,
   250→   250→    labels: [],
   251→   251→    selectedLayerId: initialDoc.layers[0]?.id ?? null,
   252→   252→    selectedLayerIds: [],
   253→   253→    selectedClusterStart: null,
   254→   254→    activeTab: "word",
   255→   255→    stressResult: null,
   256→   256→    viewScale: 0.12,
   257→   257→    favoriteFontIds: [],
   258→   258→
   259→   259→    // گفت‌وگوی افزودن/ویرایش متن — بسته به‌صورت پیش‌فرض
   260→   260→    textDialogOpen: false,
   261→   261→    editingTextLayerId: null,
   262→   262→    openTextDialog: (editLayerId) => set({ textDialogOpen: true, editingTextLayerId: editLayerId ?? null }),
   263→   263→    closeTextDialog: () => set({ textDialogOpen: false, editingTextLayerId: null }),
   264→   264→
   265→   265→    selectLayer: (id) => set({ selectedLayerId: id, selectedClusterStart: null, selectedLayerIds: [] }),
   266→   266→    selectAllLayers: () =>
   267→   267→      set({
   268→   268→        // لایه‌های قفل‌شده را نمی‌شود کشید؛ در انتخاب‌همه نمی‌آیند
   269→   269→        selectedLayerIds: get().doc.layers.filter((l) => !l.locked).map((l) => l.id),
   270→   270→        selectedClusterStart: null,
   271→   271→      }),
   272→   272→    clearMultiSelect: () => set({ selectedLayerIds: [] }),
   273→   273→    selectLetter: (clusterStart) => set({ selectedClusterStart: clusterStart }),
   274→   274→    setActiveTab: (tab) => set({ activeTab: tab }),
   275→   275→    setViewScale: (scale) => set({ viewScale: scale }),
   276→   276→    toggleFavoriteFont: (id) => {
   277→   277→      if (!PELC_FONT_IDS.has(id)) return
   278→   278→      const cur = get().favoriteFontIds
   279→   279→      const next = cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id]
   280→   280→      if (typeof window !== "undefined") {
   281→   281→        try {
   282→   282→          window.localStorage.setItem(FAVORITES_KEY, JSON.stringify(next))
   283→   283→        } catch {
   284→   284→          /* حافظه در دسترس نیست (مثلاً حالت خصوصی) — فقط در حافظهٔ نشست */
   285→   285→        }

   287→   287→      set({ favoriteFontIds: next })
   288→   288→    },
   289→   289→    hydrateFavoriteFonts: () => {
   290→   290→      if (typeof window === "undefined") return
   291→   291→      try {
   292→   292→        const raw = window.localStorage.getItem(FAVORITES_KEY)
   293→   293→        if (!raw) return
   294→   294→        const parsed: unknown = JSON.parse(raw)
   295→   295→        if (!Array.isArray(parsed)) return
   296→   296→        // فقط شناسه‌های معتبر مخزن قلم پذیرفته می‌شوند
   297→   297→        set({
   298→   298→          favoriteFontIds: parsed.filter(
   299→   299→            (x): x is string => typeof x === "string" && PELC_FONT_IDS.has(x),
   300→   300→          ),
   301→   301→        })
   302→   302→      } catch {
   303→   303→        /* دادهٔ خراب — نادیده گرفته می‌شود */
   304→   304→      }
   305→   305→    },

   311→   311→    },
   312→   312→    redo: () => {

   315→   315→      if (doc) syncFromHistory(set, get, doc)

   319→   319→      const doc = h.jumpTo(index)
   320→   320→      syncFromHistory(set, get, doc)
   321→   321→    },
   322→   322→    runStressTest: () => {
   323→   323→      // معیار پذیرش پروژه: ۵۰۰ عمل متوالی بدون خطا + واگرد تا سند آغازین
   324→   324→      const h = ensureHistory(get().doc)
   325→   325→      let doc = get().doc
   326→   326→      const layerId = doc.layers[0]?.id
   327→   327→      if (!layerId) {
   328→   328→        set({ stressResult: "سند بدون لایه است" })
   329→   329→        return
   330→   330→      }
   331→   331→      try {
   332→   332→        const t0 = performance.now()
   333→   333→        for (let i = 0; i < 500; i++) {
   334→   334→          const kashida = "\u0640".repeat((i % 3) + 1)
   335→   335→          const next = withLayer(doc, layerId, (cur) => ({
   336→   336→            ...cur,
   337→   337→            text: { ...cur.text!, text: `آزمونِ تاریخچه ${i + 1}${kashida}` },
   338→   338→          }))
   339→   339→          doc = h.apply(next, `گام آزمون ${i + 1}`)
   340→   340→        }
   341→   341→        const elapsed = Math.round(performance.now() - t0)
   342→   342→        syncFromHistory(set, get, doc)
   343→   343→        // بررسی واگرد کامل
   344→   344→        let ok = true
   345→   345→        for (let i = 0; i < 500; i++) {
   346→   346→          const d = h.undo()
   347→   347→          if (d === null) { ok = false; break }
   348→   348→        }
   349→   349→        const backToBase = h.position === 0
   350→   350→        set({
   351→   351→          stressResult: `موفق: ۵۰۰ گام در ${elapsed} میلی‌ثانیه ثبت شد (ظرفیت ${DEFAULT_CAPACITY}). واگرد کامل تا سند آغازین: ${ok && backToBase ? "بله" : "خیر"}`,
   352→   352→        })
   353→   353→        // بازگشت به آخرین وضعیت
   354→   354→        for (let i = 0; i < 500; i++) {
   355→   355→          if (!h.redo()) break
   356→   356→        }
   357→   357→        syncFromHistory(set, get, h.current)
   358→   358→      } catch (e) {
   359→   359→        set({ stressResult: `خطا: ${e instanceof Error ? e.message : String(e)}` })
   360→   360→      }
   361→   361→    },
   362→   362→
   363→   363→    commit: (next, label, coalesceKey) => {
   364→   364→      const h = ensureHistory(next)
   365→   365→      const doc = h.apply(next, label, coalesceKey ?? null)
   366→   366→      syncFromHistory(set, get, doc)
   367→   367→    },
   368→   368→    beginLiveUpdate: (next) => {
   369→   369→      liveDoc = next
   370→   370→      set({ doc: next })
   371→   371→    },
   372→   372→    commitLiveUpdate: (label) => {
   373→   373→      if (!liveDoc) return
   374→   374→      const h = ensureHistory(liveDoc)
   375→   375→      const doc = h.push(liveDoc, label)
   376→   376→      liveDoc = null
   377→   377→      syncFromHistory(set, get, doc)
   378→   378→    },
   379→   379→
   380→   380→    addTextLayer: (text) => {
   381→   381→      const state = get()
   382→   382→      const trimmed = text.trim()
   383→   383→      if (!trimmed) return null
   384→   384→      const layer = newLayer(`متن ${state.doc.layers.length + 1}`)
   385→   385→      layer.text = { ...layer.text!, text: trimmed }
   386→   386→      // جای تقریبی: میانهٔ افقی بوم و یک‌سوم بالا — کاربر با کشیدن یا تاچ‌پد دقیق می‌کند
   387→   387→      layer.transform = {
   388→   388→        tx: Math.round(state.doc.canvas.width / 2 - 320),
   389→   389→        ty: Math.round(state.doc.canvas.height / 2 - 120),
   390→   390→        scaleX: 1,
   391→   391→        scaleY: 1,
   392→   392→        rotationDeg: 0,
   393→   393→      }
   394→   394→      const next: PelcDocument = { ...state.doc, layers: [...state.doc.layers, layer] }

   774→   774→      const start = state.selectedClusterStart
   775→   775→      if (!layer?.text || start === null) return
   776→   776→      const cluster = letterClusters(layer.text.text).find((c) => c.start === start)
   777→   777→      if (!cluster) return

   779→   779→      const mutation = insertKashidaBefore(layer.text.text, start)
   780→   780→      if (!mutation) return
   781→   781→      commitTextMutation(mutation, "درج کشیده")
   782→   782→    },
   783→   783→    removeKashida: () => {
   784→   784→      const state = get()
   785→   785→      const layer = findLayer(state.doc, state.selectedLayerId)
   786→   786→      const start = state.selectedClusterStart

   789→   789→      if (!mutation) return
   790→   790→      commitTextMutation(mutation, "حذف کشیده")
   791→   791→    },
   792→   792→    addMark: (type) => {
   793→   793→      const state = get()
   794→   794→      const layer = findLayer(state.doc, state.selectedLayerId)
   795→   795→      const start = state.selectedClusterStart
   796→   796→      if (!layer?.text || start === null) return
   797→   797→      const mutation = insertMarkAfter(layer.text.text, start, markChar(type))
   798→   798→      if (!mutation) return
   799→   799→      // رکورد تنظیم علامت در perLetter.marks ثبت می‌شود (سازگار با مدل اندروید)
   800→   800→      const key = String(start)
   801→   801→      const existing = layer.text.perLetter[key] ?? defaultLetterOverride()
   802→   802→      const perLetter0 = {
   803→   803→        ...layer.text.perLetter,
   804→   804→        [key]: { ...existing, marks: [...existing.marks, { type, dx: 0, dy: 0, scale: 1 }] },
   805→   805→      }
   806→   806→      commitTextMutation(mutation, `افزودن ${markTitle(type)}`, perLetter0)
   807→   807→    },
   808→   808→    removeMark: (markIndex) => {
   809→   809→      const state = get()
   810→   810→      const layer = findLayer(state.doc, state.selectedLayerId)
   811→   811→      const start = state.selectedClusterStart
   812→   812→      if (!layer?.text || start === null) return
   813→   813→      const mutation = removeMarkAt(layer.text.text, start, markIndex)
   814→   814→      if (!mutation) return
   815→   815→      const key = String(start)
   816→   816→      const existing = layer.text.perLetter[key]
   817→   817→      const perLetter0 = { ...layer.text.perLetter }
   818→   818→      if (existing) {

   821→   821→        else perLetter0[key] = updated
   822→   822→      }
   823→   823→      commitTextMutation(mutation, "حذف علامت", perLetter0)
   824→   824→    },
   825→   825→
   826→   826→    clusters: () => {

   828→   828→      const layer = findLayer(state.doc, state.selectedLayerId)
