   148→/** تاریخچه بیرون از استور نگه داشته می‌شود (عکس‌فوری‌ها عمق‌نمایی نمی‌خواهند) */

   152→function ensureHistory(initial: PelcDocument): EditorHistory {
   153→  if (!history) history = new EditorHistory(initial, DEFAULT_CAPACITY)
   154→  return history
   155→}

   157→/** اعمال سند تازه از تاریخچه + حفظ انتخاب‌های معتبر */
   158→function syncFromHistory(
   159→  set: (partial: Partial<EditorState>) => void,
   160→  get: () => EditorState,
   161→  doc: PelcDocument,
   162→) {
   163→  const h = history!
   164→  const sel = get().selectedLayerId
   165→  const layer = doc.layers.some((l) => l.id === sel) ? sel : (doc.layers[0]?.id ?? null)
   166→  // حفظ انتخاب چندگانهٔ معتبر — شناسه‌های حذف‌شده کنار می‌روند
   167→  const multi = get().selectedLayerIds.filter((id) => doc.layers.some((l) => l.id === id))
   168→  const keepStart = get().selectedClusterStart
   169→  let nextStart = keepStart
   170→  if (keepStart !== null) {
   171→    const l = doc.layers.find((x) => x.id === layer)
   172→    const text = l?.text?.text ?? ""
   173→    if (!l?.text || !letterClusters(text).some((c) => c.start === keepStart)) nextStart = null
   174→  }
   175→  set({
   176→    doc,
   177→    historyVersion: h.version,
   178→    stepCount: h.stepCount,
   179→    position: h.position,
   180→    canUndo: h.canUndo,
   181→    canRedo: h.canRedo,
   182→    labels: h.labels(),
   183→    selectedLayerId: layer,
   184→    selectedLayerIds: multi,
   185→    selectedClusterStart: nextStart,
   186→  })
   187→}

   217→export const useEditorStore = create<EditorState>((set, get) => {
   218→  /** اعمال جهش متن با بازنگاشت perLetter و پیگیری انتخاب حرف */
   219→  const commitTextMutation = (mutation: TextMutation, label: string, extraPerLetter?: Record<string, LetterOverride>) => {
   220→    const state = get()
   221→    const layer = findLayer(state.doc, state.selectedLayerId)
   222→    if (!layer?.text) return
   223→    const base = extraPerLetter ?? layer.text.perLetter
   224→    const perLetter = remapPerLetter(base, mutation)
   225→    const next = withLayer(state.doc, layer.id, (l) => ({
   226→      ...l,
   227→      text: { ...l.text!, text: mutation.newText, perLetter },
   228→    }))
   229→    const keepStart = state.selectedClusterStart
   230→    state.commit(next, label)
   231→    if (keepStart !== null) {
   232→      const mapped = keepStart >= mutation.shiftFrom ? keepStart + mutation.delta : keepStart
   233→      const clusters = letterClusters(mutation.newText)
   234→      if (clusters.some((c) => c.start === mapped)) {
   235→        set({ selectedClusterStart: mapped })
   236→      }
   237→    }
   238→  }
   239→
   240→  const initialDoc = newDocument()
   241→  history = new EditorHistory(initialDoc, DEFAULT_CAPACITY)
   242→
   243→  return {
   244→    doc: initialDoc,
   245→    historyVersion: 0,
   246→    stepCount: 0,
   247→    position: 0,
   248→    canUndo: false,
   249→    canRedo: false,
   250→    labels: [],
   251→    selectedLayerId: initialDoc.layers[0]?.id ?? null,
   252→    selectedLayerIds: [],
   253→    selectedClusterStart: null,
   254→    activeTab: "word",
   255→    stressResult: null,
   256→    viewScale: 0.12,
   257→    favoriteFontIds: [],
   258→
   259→    // گفت‌وگوی افزودن/ویرایش متن — بسته به‌صورت پیش‌فرض
   260→    textDialogOpen: false,
   261→    editingTextLayerId: null,
   262→    openTextDialog: (editLayerId) => set({ textDialogOpen: true, editingTextLayerId: editLayerId ?? null }),
   263→    closeTextDialog: () => set({ textDialogOpen: false, editingTextLayerId: null }),
   264→
   265→    selectLayer: (id) => set({ selectedLayerId: id, selectedClusterStart: null, selectedLayerIds: [] }),
   266→    selectAllLayers: () =>
   267→      set({
   268→        // لایه‌های قفل‌شده را نمی‌شود کشید؛ در انتخاب‌همه نمی‌آیند
   269→        selectedLayerIds: get().doc.layers.filter((l) => !l.locked).map((l) => l.id),
   270→        selectedClusterStart: null,
   271→      }),
   272→    clearMultiSelect: () => set({ selectedLayerIds: [] }),
   273→    selectLetter: (clusterStart) => set({ selectedClusterStart: clusterStart }),
   274→    setActiveTab: (tab) => set({ activeTab: tab }),
   275→    setViewScale: (scale) => set({ viewScale: scale }),
   276→    toggleFavoriteFont: (id) => {
   277→      if (!PELC_FONT_IDS.has(id)) return
   278→      const cur = get().favoriteFontIds
   279→      const next = cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id]
   280→      if (typeof window !== "undefined") {
   281→        try {
   282→          window.localStorage.setItem(FAVORITES_KEY, JSON.stringify(next))
   283→        } catch {
   284→          /* حافظه در دسترس نیست (مثلاً حالت خصوصی) — فقط در حافظهٔ نشست */
   285→        }
   286→      }
   287→      set({ favoriteFontIds: next })
   288→    },
   289→    hydrateFavoriteFonts: () => {
   290→      if (typeof window === "undefined") return
   291→      try {
   292→        const raw = window.localStorage.getItem(FAVORITES_KEY)
   293→        if (!raw) return
   294→        const parsed: unknown = JSON.parse(raw)
   295→        if (!Array.isArray(parsed)) return
   296→        // فقط شناسه‌های معتبر مخزن قلم پذیرفته می‌شوند
   297→        set({
   298→          favoriteFontIds: parsed.filter(
   299→            (x): x is string => typeof x === "string" && PELC_FONT_IDS.has(x),
   300→          ),
   301→        })
   302→      } catch {
   303→        /* دادهٔ خراب — نادیده گرفته می‌شود */
   304→      }
   305→    },
   306→
   307→    undo: () => {
   308→      const h = ensureHistory(get().doc)
   309→      const doc = h.undo()
   310→      if (doc) syncFromHistory(set, get, doc)
   311→    },
   312→    redo: () => {
   313→      const h = ensureHistory(get().doc)
   314→      const doc = h.redo()
   315→      if (doc) syncFromHistory(set, get, doc)
   316→    },
   317→    jumpTo: (index) => {
   318→      const h = ensureHistory(get().doc)
   319→      const doc = h.jumpTo(index)
   320→      syncFromHistory(set, get, doc)
   321→    },
   322→    runStressTest: () => {
   323→      // معیار پذیرش پروژه: ۵۰۰ عمل متوالی بدون خطا + واگرد تا سند آغازین
   324→      const h = ensureHistory(get().doc)
   325→      let doc = get().doc
   326→      const layerId = doc.layers[0]?.id
   327→      if (!layerId) {
   328→        set({ stressResult: "سند بدون لایه است" })
   329→        return
   330→      }
   331→      try {
   332→        const t0 = performance.now()
   333→        for (let i = 0; i < 500; i++) {
   334→          const kashida = "\u0640".repeat((i % 3) + 1)
   335→          const next = withLayer(doc, layerId, (cur) => ({
   336→            ...cur,
   337→            text: { ...cur.text!, text: `آزمونِ تاریخچه ${i + 1}${kashida}` },
   338→          }))
   339→          doc = h.apply(next, `گام آزمون ${i + 1}`)
   340→        }
   341→        const elapsed = Math.round(performance.now() - t0)
   342→        syncFromHistory(set, get, doc)
   343→        // بررسی واگرد کامل
   344→        let ok = true
   345→        for (let i = 0; i < 500; i++) {
   346→          const d = h.undo()
   347→          if (d === null) { ok = false; break }
   348→        }
   349→        const backToBase = h.position === 0
   350→        set({
   351→          stressResult: `موفق: ۵۰۰ گام در ${elapsed} میلی‌ثانیه ثبت شد (ظرفیت ${DEFAULT_CAPACITY}). واگرد کامل تا سند آغازین: ${ok && backToBase ? "بله" : "خیر"}`,
   352→        })
   353→        // بازگشت به آخرین وضعیت
   354→        for (let i = 0; i < 500; i++) {
   355→          if (!h.redo()) break
   356→        }
   357→        syncFromHistory(set, get, h.current)
   358→      } catch (e) {
   359→        set({ stressResult: `خطا: ${e instanceof Error ? e.message : String(e)}` })
   360→      }
   361→    },
   362→
   363→    commit: (next, label, coalesceKey) => {
   364→      const h = ensureHistory(next)
   365→      const doc = h.apply(next, label, coalesceKey ?? null)
   366→      syncFromHistory(set, get, doc)
   367→    },
   368→    beginLiveUpdate: (next) => {
   369→      liveDoc = next
   370→      set({ doc: next })
   371→    },
   372→    commitLiveUpdate: (label) => {
   373→      if (!liveDoc) return
   374→      const h = ensureHistory(liveDoc)
   375→      const doc = h.push(liveDoc, label)
   376→      liveDoc = null
   377→      syncFromHistory(set, get, doc)
   378→    },
   379→
   380→    addTextLayer: (text) => {
   381→      const state = get()
   382→      const trimmed = text.trim()
   383→      if (!trimmed) return null
   384→      const layer = newLayer(`متن ${state.doc.layers.length + 1}`)
   385→      layer.text = { ...layer.text!, text: trimmed }
   386→      // جای تقریبی: میانهٔ افقی بوم و یک‌سوم بالا — کاربر با کشیدن یا تاچ‌پد دقیق می‌کند
   387→      layer.transform = {
   388→        tx: Math.round(state.doc.canvas.width / 2 - 320),
   389→        ty: Math.round(state.doc.canvas.height / 2 - 120),
   390→        scaleX: 1,
   391→        scaleY: 1,
   392→        rotationDeg: 0,
   393→      }
   394→      const next: PelcDocument = { ...state.doc, layers: [...state.doc.layers, layer] }
   395→      state.commit(next, "افزودن متن")
   396→      set({ selectedLayerId: layer.id, selectedClusterStart: null, selectedLayerIds: [], activeTab: "word" })
   397→      return layer.id
   398→    },
   399→    addImageLayer: (assetId, width, height) => {
   400→      const state = get()
   401→      const layer = newLayer("عکس")
   402→      layer.kind = "IMAGE"
   403→      layer.text = null
   404→      layer.image = { assetId, width, height }
   405→      // پوشش کامل بوم (cover) وسط‌چین — متن‌های آماده روی آن می‌نشینند؛
   406→      // کاربر با کشیدن جابه‌جا و با دستگیره‌های گوشه بزرگ/کوچک می‌کند.
   407→      const s = Math.max(state.doc.canvas.width / width, state.doc.canvas.height / height)
   408→      layer.transform = {
   409→        tx: Math.round((state.doc.canvas.width - width * s) / 2),
   410→        ty: Math.round((state.doc.canvas.height - height * s) / 2),
   411→        scaleX: s,
   412→        scaleY: s,
   413→        rotationDeg: 0,
   414→      }
   415→      // پایین‌ترین ترتیب — زیر همهٔ لایه‌های متنی تا متن روی عکس بیفتد
   416→      const next: PelcDocument = { ...state.doc, layers: [layer, ...state.doc.layers] }
   417→      state.commit(next, "افزودن عکس")
   418→      set({ selectedLayerId: layer.id, selectedClusterStart: null, selectedLayerIds: [] })
   419→    },
   420→    deleteLayer: (id) => {
   421→      const state = get()
   422→      if (state.doc.layers.length <= 1) return
   423→      const next: PelcDocument = { ...state.doc, layers: state.doc.layers.filter((l) => l.id !== id) }
   424→      state.commit(next, "حذف لایه")
   425→      set({ selectedLayerIds: state.selectedLayerIds.filter((x) => x !== id) })
   426→    },
   427→    duplicateLayer: (id) => {
   428→      const state = get()
   429→      const src = state.doc.layers.find((l) => l.id === id)
   430→      if (!src) return
   431→      const copy: Layer = JSON.parse(JSON.stringify(src))
   432→      copy.id = `${src.id}-copy-${Date.now()}`
   433→      copy.name = `${src.name} (کپی)`
   434→      const idx = state.doc.layers.findIndex((l) => l.id === id)
   435→      const layers = [...state.doc.layers]
   436→      layers.splice(idx + 1, 0, copy)
   437→      state.commit({ ...state.doc, layers }, "تکثیر لایه")
   438→      set({ selectedLayerId: copy.id, selectedClusterStart: null, selectedLayerIds: [] })
   439→    },
   440→
   441→    splitLayerIntoWords: (id, opts) => {
   442→      const state = get()
   443→      const layer = findLayer(state.doc, id)
   444→      if (!layer?.text) return false
   445→      const tokens = tokenizeWords(layer.text.text)
   446→      if (tokens.length < 2) return false
   447→      const boxes = measureWordPositions(layer, state.viewScale)
   448→      if (!boxes) return false
   449→
   450→      const wordLayers: Layer[] = tokens.map((tk, i) => {
   451→        const copy: Layer = JSON.parse(JSON.stringify(layer))
   452→        copy.id = newId()
   453→        copy.name = `کلمه: ${tk.token.slice(0, 16)}`
   454→        copy.text = {
   455→          ...layer.text!,
   456→          text: tk.token,
   457→          perLetter: slicePerLetter(layer.text!.perLetter, tk.rawStart, tk.token.length),
