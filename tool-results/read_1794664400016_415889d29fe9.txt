   139→   139→function ensureHistory(initial: PelcDocument): EditorHistory {
   140→   140→  if (!history) history = new EditorHistory(initial, DEFAULT_CAPACITY)
   141→   141→  return history

   144→   144→/** اعمال سند تازه از تاریخچه + حفظ انتخاب‌های معتبر */
   145→   145→function syncFromHistory(
   146→   146→  set: (partial: Partial<EditorState>) => void,
   147→   147→  get: () => EditorState,
   148→   148→  doc: PelcDocument,
   149→   149→) {
   150→   150→  const h = history!
   151→   151→  const sel = get().selectedLayerId
   152→   152→  const layer = doc.layers.some((l) => l.id === sel) ? sel : (doc.layers[0]?.id ?? null)
   153→   153→  // حفظ انتخاب چندگانهٔ معتبر — شناسه‌های حذف‌شده کنار می‌روند
   154→   154→  const multi = get().selectedLayerIds.filter((id) => doc.layers.some((l) => l.id === id))
   155→   155→  const keepStart = get().selectedClusterStart
   156→   156→  let nextStart = keepStart
   157→   157→  if (keepStart !== null) {
   158→   158→    const l = doc.layers.find((x) => x.id === layer)
   159→   159→    const text = l?.text?.text ?? ""
   160→   160→    if (!l?.text || !letterClusters(text).some((c) => c.start === keepStart)) nextStart = null
   161→   161→  }
   162→   162→  set({
   163→   163→    doc,
   164→   164→    historyVersion: h.version,
   165→   165→    stepCount: h.stepCount,
   166→   166→    position: h.position,
   167→   167→    canUndo: h.canUndo,
   168→   168→    canRedo: h.canRedo,
   169→   169→    labels: h.labels(),
   170→   170→    selectedLayerId: layer,
   171→   171→    selectedLayerIds: multi,
   172→   172→    selectedClusterStart: nextStart,
   173→   173→  })
   174→   174→}

   204→   204→export const useEditorStore = create<EditorState>((set, get) => {
   205→   205→  /** اعمال جهش متن با بازنگاشت perLetter و پیگیری انتخاب حرف */
   206→   206→  const commitTextMutation = (mutation: TextMutation, label: string, extraPerLetter?: Record<string, LetterOverride>) => {
   207→   207→    const state = get()
   208→   208→    const layer = findLayer(state.doc, state.selectedLayerId)
   209→   209→    if (!layer?.text) return
   210→   210→    const base = extraPerLetter ?? layer.text.perLetter
   211→   211→    const perLetter = remapPerLetter(base, mutation)
   212→   212→    const next = withLayer(state.doc, layer.id, (l) => ({
   213→   213→      ...l,
   214→   214→      text: { ...l.text!, text: mutation.newText, perLetter },
   215→   215→    }))
   216→   216→    const keepStart = state.selectedClusterStart
   217→   217→    state.commit(next, label)
   218→   218→    if (keepStart !== null) {
   219→   219→      const mapped = keepStart >= mutation.shiftFrom ? keepStart + mutation.delta : keepStart
   220→   220→      const clusters = letterClusters(mutation.newText)
   221→   221→      if (clusters.some((c) => c.start === mapped)) {
   222→   222→        set({ selectedClusterStart: mapped })
   223→   223→      }
   224→   224→    }
   225→   225→  }
   226→   226→
   227→   227→  const initialDoc = newDocument()
   228→   228→  history = new EditorHistory(initialDoc, DEFAULT_CAPACITY)
   229→   229→
   230→   230→  return {
   231→   231→    doc: initialDoc,
   232→   232→    historyVersion: 0,
   233→   233→    stepCount: 0,
   234→   234→    position: 0,
   235→   235→    canUndo: false,
   236→   236→    canRedo: false,
   237→   237→    labels: [],
   238→   238→    selectedLayerId: initialDoc.layers[0]?.id ?? null,
   239→   239→    selectedLayerIds: [],
   240→   240→    selectedClusterStart: null,
   241→   241→    activeTab: "word",
   242→   242→    stressResult: null,
   243→   243→    viewScale: 0.12,
   244→   244→    favoriteFontIds: [],
   245→   245→
   246→   246→    selectLayer: (id) => set({ selectedLayerId: id, selectedClusterStart: null, selectedLayerIds: [] }),
   247→   247→    selectAllLayers: () =>
   248→   248→      set({
   249→   249→        // لایه‌های قفل‌شده را نمی‌شود کشید؛ در انتخاب‌همه نمی‌آیند
   250→   250→        selectedLayerIds: get().doc.layers.filter((l) => !l.locked).map((l) => l.id),
   251→   251→        selectedClusterStart: null,
   252→   252→      }),
   253→   253→    clearMultiSelect: () => set({ selectedLayerIds: [] }),
   254→   254→    selectLetter: (clusterStart) => set({ selectedClusterStart: clusterStart }),
   255→   255→    setActiveTab: (tab) => set({ activeTab: tab }),
   256→   256→    setViewScale: (scale) => set({ viewScale: scale }),
   257→   257→    toggleFavoriteFont: (id) => {
   258→   258→      if (!PELC_FONT_IDS.has(id)) return
   259→   259→      const cur = get().favoriteFontIds
   260→   260→      const next = cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id]
   261→   261→      if (typeof window !== "undefined") {
   262→   262→        try {
   263→   263→          window.localStorage.setItem(FAVORITES_KEY, JSON.stringify(next))
   264→   264→        } catch {
   265→   265→          /* حافظه در دسترس نیست (مثلاً حالت خصوصی) — فقط در حافظهٔ نشست */
   266→   266→        }
   267→   267→      }
   268→   268→      set({ favoriteFontIds: next })
   269→   269→    },
   270→   270→    hydrateFavoriteFonts: () => {
   271→   271→      if (typeof window === "undefined") return
   272→   272→      try {
   273→   273→        const raw = window.localStorage.getItem(FAVORITES_KEY)
   274→   274→        if (!raw) return
   275→   275→        const parsed: unknown = JSON.parse(raw)
   276→   276→        if (!Array.isArray(parsed)) return
   277→   277→        // فقط شناسه‌های معتبر مخزن قلم پذیرفته می‌شوند
   278→   278→        set({
   279→   279→          favoriteFontIds: parsed.filter(
   280→   280→            (x): x is string => typeof x === "string" && PELC_FONT_IDS.has(x),
   281→   281→          ),
   282→   282→        })
   283→   283→      } catch {
   284→   284→        /* دادهٔ خراب — نادیده گرفته می‌شود */
   285→   285→      }
   286→   286→    },
   287→   287→
   288→   288→    undo: () => {
   289→   289→      const h = ensureHistory(get().doc)
   290→   290→      const doc = h.undo()
   291→   291→      if (doc) syncFromHistory(set, get, doc)
   292→   292→    },
   293→   293→    redo: () => {
   294→   294→      const h = ensureHistory(get().doc)
   295→   295→      const doc = h.redo()
   296→   296→      if (doc) syncFromHistory(set, get, doc)
   297→   297→    },
   298→   298→    jumpTo: (index) => {
   299→   299→      const h = ensureHistory(get().doc)
   300→   300→      const doc = h.jumpTo(index)
   301→   301→      syncFromHistory(set, get, doc)
   302→   302→    },
   303→   303→    runStressTest: () => {
   304→   304→      // معیار پذیرش پروژه: ۵۰۰ عمل متوالی بدون خطا + واگرد تا سند آغازین
   305→   305→      const h = ensureHistory(get().doc)
   306→   306→      let doc = get().doc
   307→   307→      const layerId = doc.layers[0]?.id
   308→   308→      if (!layerId) {
   309→   309→        set({ stressResult: "سند بدون لایه است" })
   310→   310→        return
   311→   311→      }
   312→   312→      try {
   313→   313→        const t0 = performance.now()
   314→   314→        for (let i = 0; i < 500; i++) {
   315→   315→          const kashida = "\u0640".repeat((i % 3) + 1)
   316→   316→          const next = withLayer(doc, layerId, (cur) => ({
   317→   317→            ...cur,
   318→   318→            text: { ...cur.text!, text: `آزمونِ تاریخچه ${i + 1}${kashida}` },
   319→   319→          }))
   320→   320→          doc = h.apply(next, `گام آزمون ${i + 1}`)
   321→   321→        }
   322→   322→        const elapsed = Math.round(performance.now() - t0)
   323→   323→        syncFromHistory(set, get, doc)
   324→   324→        // بررسی واگرد کامل
   325→   325→        let ok = true
   326→   326→        for (let i = 0; i < 500; i++) {
   327→   327→          const d = h.undo()
   328→   328→          if (d === null) { ok = false; break }
   329→   329→        }
   330→   330→        const backToBase = h.position === 0
   331→   331→        set({
   332→   332→          stressResult: `موفق: ۵۰۰ گام در ${elapsed} میلی‌ثانیه ثبت شد (ظرفیت ${DEFAULT_CAPACITY}). واگرد کامل تا سند آغازین: ${ok && backToBase ? "بله" : "خیر"}`,
   333→   333→        })
   334→   334→        // بازگشت به آخرین وضعیت
   335→   335→        for (let i = 0; i < 500; i++) {
   336→   336→          if (!h.redo()) break
   337→   337→        }
   338→   338→        syncFromHistory(set, get, h.current)
   339→   339→      } catch (e) {
   340→   340→        set({ stressResult: `خطا: ${e instanceof Error ? e.message : String(e)}` })
   341→   341→      }
   342→   342→    },
   343→   343→
   344→   344→    commit: (next, label, coalesceKey) => {
   345→   345→      const h = ensureHistory(next)
   346→   346→      const doc = h.apply(next, label, coalesceKey ?? null)
   347→   347→      syncFromHistory(set, get, doc)
   348→   348→    },
   349→   349→    beginLiveUpdate: (next) => {
   350→   350→      liveDoc = next
   351→   351→      set({ doc: next })
   352→   352→    },
   353→   353→    commitLiveUpdate: (label) => {
   354→   354→      if (!liveDoc) return
   355→   355→      const h = ensureHistory(liveDoc)
   356→   356→      const doc = h.push(liveDoc, label)
   357→   357→      liveDoc = null
   358→   358→      syncFromHistory(set, get, doc)
   359→   359→    },
   360→   360→
   361→   361→    addLayer: () => {
   362→   362→      const state = get()
   363→   363→      const layer = newLayer(`لایهٔ ${state.doc.layers.length + 1}`)
   364→   364→      const next: PelcDocument = { ...state.doc, layers: [...state.doc.layers, layer] }
   365→   365→      state.commit(next, "افزودن لایه")
   366→   366→      set({ selectedLayerId: layer.id, selectedClusterStart: null, selectedLayerIds: [], activeTab: "word" })
   367→   367→    },
   368→   368→    deleteLayer: (id) => {
   369→   369→      const state = get()
   370→   370→      if (state.doc.layers.length <= 1) return
   371→   371→      const next: PelcDocument = { ...state.doc, layers: state.doc.layers.filter((l) => l.id !== id) }
   372→   372→      state.commit(next, "حذف لایه")
   373→   373→      set({ selectedLayerIds: state.selectedLayerIds.filter((x) => x !== id) })
   374→   374→    },
   375→   375→    duplicateLayer: (id) => {
   376→   376→      const state = get()
   377→   377→      const src = state.doc.layers.find((l) => l.id === id)
   378→   378→      if (!src) return
   379→   379→      const copy: Layer = JSON.parse(JSON.stringify(src))
   380→   380→      copy.id = `${src.id}-copy-${Date.now()}`
   381→   381→      copy.name = `${src.name} (کپی)`
   382→   382→      const idx = state.doc.layers.findIndex((l) => l.id === id)
   383→   383→      const layers = [...state.doc.layers]
   384→   384→      layers.splice(idx + 1, 0, copy)
   385→   385→      state.commit({ ...state.doc, layers }, "تکثیر لایه")
   386→   386→      set({ selectedLayerId: copy.id, selectedClusterStart: null, selectedLayerIds: [] })
   387→   387→    },
   388→   388→
   389→   389→    splitLayerIntoWords: (id) => {
   390→   390→      const state = get()
   391→   391→      const layer = findLayer(state.doc, id)
   392→   392→      if (!layer?.text) return false
   393→   393→      const tokens = tokenizeWords(layer.text.text)
   394→   394→      if (tokens.length < 2) return false
   395→   395→      const boxes = measureWordPositions(layer, state.viewScale)
   396→   396→      if (!boxes) return false
   397→   397→
   398→   398→      const wordLayers: Layer[] = tokens.map((tk, i) => {
   399→   399→        const copy: Layer = JSON.parse(JSON.stringify(layer))
   400→   400→        copy.id = newId()
   401→   401→        copy.name = `کلمه: ${tk.token.slice(0, 16)}`
   402→   402→        copy.text = {
   403→   403→          ...layer.text!,
   404→   404→          text: tk.token,
   405→   405→          perLetter: slicePerLetter(layer.text!.perLetter, tk.rawStart, tk.token.length),
   406→   406→        }
   407→   407→        copy.transform = {
   408→   408→          ...layer.transform,
   409→   409→          tx: Math.round(layer.transform.tx + boxes[i].x),
   410→   410→          ty: Math.round(layer.transform.ty + boxes[i].y),
   411→   411→        }
   412→   412→        return copy
