   135→/** تاریخچه بیرون از استور نگه داشته می‌شود (عکس‌فوری‌ها عمق‌نمایی نمی‌خواهند) */
   136→let history: EditorHistory | null = null

   139→function ensureHistory(initial: PelcDocument): EditorHistory {
   140→  if (!history) history = new EditorHistory(initial, DEFAULT_CAPACITY)
   141→  return history
   142→}
   143→
   144→/** اعمال سند تازه از تاریخچه + حفظ انتخاب‌های معتبر */
   145→function syncFromHistory(
   146→  set: (partial: Partial<EditorState>) => void,
   147→  get: () => EditorState,
   148→  doc: PelcDocument,
   149→) {
   150→  const h = history!
   151→  const sel = get().selectedLayerId
   152→  const layer = doc.layers.some((l) => l.id === sel) ? sel : (doc.layers[0]?.id ?? null)
   153→  // حفظ انتخاب چندگانهٔ معتبر — شناسه‌های حذف‌شده کنار می‌روند
   154→  const multi = get().selectedLayerIds.filter((id) => doc.layers.some((l) => l.id === id))
   155→  const keepStart = get().selectedClusterStart
   156→  let nextStart = keepStart
   157→  if (keepStart !== null) {
   158→    const l = doc.layers.find((x) => x.id === layer)
   159→    const text = l?.text?.text ?? ""
   160→    if (!l?.text || !letterClusters(text).some((c) => c.start === keepStart)) nextStart = null
   161→  }
   162→  set({
   163→    doc,
   164→    historyVersion: h.version,
   165→    stepCount: h.stepCount,
   166→    position: h.position,
   167→    canUndo: h.canUndo,
   168→    canRedo: h.canRedo,
   169→    labels: h.labels(),
   170→    selectedLayerId: layer,
   171→    selectedLayerIds: multi,
   172→    selectedClusterStart: nextStart,
   173→  })
   174→}

   204→export const useEditorStore = create<EditorState>((set, get) => {
   205→  /** اعمال جهش متن با بازنگاشت perLetter و پیگیری انتخاب حرف */
   206→  const commitTextMutation = (mutation: TextMutation, label: string, extraPerLetter?: Record<string, LetterOverride>) => {
   207→    const state = get()
   208→    const layer = findLayer(state.doc, state.selectedLayerId)
   209→    if (!layer?.text) return
   210→    const base = extraPerLetter ?? layer.text.perLetter
   211→    const perLetter = remapPerLetter(base, mutation)
   212→    const next = withLayer(state.doc, layer.id, (l) => ({
   213→      ...l,
   214→      text: { ...l.text!, text: mutation.newText, perLetter },
   215→    }))
   216→    const keepStart = state.selectedClusterStart
   217→    state.commit(next, label)
   218→    if (keepStart !== null) {
   219→      const mapped = keepStart >= mutation.shiftFrom ? keepStart + mutation.delta : keepStart
   220→      const clusters = letterClusters(mutation.newText)
   221→      if (clusters.some((c) => c.start === mapped)) {
   222→        set({ selectedClusterStart: mapped })
   223→      }
   224→    }
   225→  }
   226→
   227→  const initialDoc = newDocument()
   228→  history = new EditorHistory(initialDoc, DEFAULT_CAPACITY)
   229→
   230→  return {
   231→    doc: initialDoc,
   232→    historyVersion: 0,
   233→    stepCount: 0,
   234→    position: 0,
   235→    canUndo: false,
   236→    canRedo: false,
   237→    labels: [],
   238→    selectedLayerId: initialDoc.layers[0]?.id ?? null,
   239→    selectedLayerIds: [],
   240→    selectedClusterStart: null,
   241→    activeTab: "word",
   242→    stressResult: null,
   243→    viewScale: 0.12,
   244→    favoriteFontIds: [],
   245→
   246→    selectLayer: (id) => set({ selectedLayerId: id, selectedClusterStart: null, selectedLayerIds: [] }),
   247→    selectAllLayers: () =>
   248→      set({
   249→        // لایه‌های قفل‌شده را نمی‌شود کشید؛ در انتخاب‌همه نمی‌آیند
   250→        selectedLayerIds: get().doc.layers.filter((l) => !l.locked).map((l) => l.id),
   251→        selectedClusterStart: null,
   252→      }),
   253→    clearMultiSelect: () => set({ selectedLayerIds: [] }),
   254→    selectLetter: (clusterStart) => set({ selectedClusterStart: clusterStart }),
   255→    setActiveTab: (tab) => set({ activeTab: tab }),
   256→    setViewScale: (scale) => set({ viewScale: scale }),
   257→    toggleFavoriteFont: (id) => {
   258→      if (!PELC_FONT_IDS.has(id)) return
   259→      const cur = get().favoriteFontIds
   260→      const next = cur.includes(id) ? cur.filter((x) => x !== id) : [...cur, id]
   261→      if (typeof window !== "undefined") {
   262→        try {
   263→          window.localStorage.setItem(FAVORITES_KEY, JSON.stringify(next))
   264→        } catch {
   265→          /* حافظه در دسترس نیست (مثلاً حالت خصوصی) — فقط در حافظهٔ نشست */
   266→        }
   267→      }
   268→      set({ favoriteFontIds: next })
   269→    },
   270→    hydrateFavoriteFonts: () => {
   271→      if (typeof window === "undefined") return
   272→      try {
   273→        const raw = window.localStorage.getItem(FAVORITES_KEY)
   274→        if (!raw) return
   275→        const parsed: unknown = JSON.parse(raw)
   276→        if (!Array.isArray(parsed)) return
   277→        // فقط شناسه‌های معتبر مخزن قلم پذیرفته می‌شوند
   278→        set({
   279→          favoriteFontIds: parsed.filter(
   280→            (x): x is string => typeof x === "string" && PELC_FONT_IDS.has(x),
   281→          ),
   282→        })
   283→      } catch {
   284→        /* دادهٔ خراب — نادیده گرفته می‌شود */
   285→      }
   286→    },
   287→
   288→    undo: () => {
   289→      const h = ensureHistory(get().doc)
   290→      const doc = h.undo()
   291→      if (doc) syncFromHistory(set, get, doc)
   292→    },
   293→    redo: () => {
   294→      const h = ensureHistory(get().doc)
   295→      const doc = h.redo()
   296→      if (doc) syncFromHistory(set, get, doc)
   297→    },
   298→    jumpTo: (index) => {
   299→      const h = ensureHistory(get().doc)
   300→      const doc = h.jumpTo(index)
   301→      syncFromHistory(set, get, doc)
   302→    },
   303→    runStressTest: () => {
   304→      // معیار پذیرش پروژه: ۵۰۰ عمل متوالی بدون خطا + واگرد تا سند آغازین
   305→      const h = ensureHistory(get().doc)
   306→      let doc = get().doc
   307→      const layerId = doc.layers[0]?.id
   308→      if (!layerId) {
   309→        set({ stressResult: "سند بدون لایه است" })
   310→        return
   311→      }
   312→      try {
   313→        const t0 = performance.now()
   314→        for (let i = 0; i < 500; i++) {
   315→          const kashida = "\u0640".repeat((i % 3) + 1)
   316→          const next = withLayer(doc, layerId, (cur) => ({
   317→            ...cur,
   318→            text: { ...cur.text!, text: `آزمونِ تاریخچه ${i + 1}${kashida}` },
   319→          }))
   320→          doc = h.apply(next, `گام آزمون ${i + 1}`)
   321→        }
   322→        const elapsed = Math.round(performance.now() - t0)
   323→        syncFromHistory(set, get, doc)
   324→        // بررسی واگرد کامل
   325→        let ok = true
   326→        for (let i = 0; i < 500; i++) {
   327→          const d = h.undo()
   328→          if (d === null) { ok = false; break }
   329→        }
   330→        const backToBase = h.position === 0
   331→        set({
   332→          stressResult: `موفق: ۵۰۰ گام در ${elapsed} میلی‌ثانیه ثبت شد (ظرفیت ${DEFAULT_CAPACITY}). واگرد کامل تا سند آغازین: ${ok && backToBase ? "بله" : "خیر"}`,
   333→        })
   334→        // بازگشت به آخرین وضعیت
   335→        for (let i = 0; i < 500; i++) {
   336→          if (!h.redo()) break
   337→        }
   338→        syncFromHistory(set, get, h.current)
   339→      } catch (e) {
   340→        set({ stressResult: `خطا: ${e instanceof Error ? e.message : String(e)}` })
   341→      }
   342→    },
   343→
   344→    commit: (next, label, coalesceKey) => {
   345→      const h = ensureHistory(next)
   346→      const doc = h.apply(next, label, coalesceKey ?? null)
   347→      syncFromHistory(set, get, doc)
   348→    },
   349→    beginLiveUpdate: (next) => {
   350→      liveDoc = next
   351→      set({ doc: next })
   352→    },
   353→    commitLiveUpdate: (label) => {
   354→      if (!liveDoc) return
   355→      const h = ensureHistory(liveDoc)
   356→      const doc = h.push(liveDoc, label)
   357→      liveDoc = null
   358→      syncFromHistory(set, get, doc)
   359→    },
   360→
   361→    addLayer: () => {
   362→      const state = get()
   363→      const layer = newLayer(`لایهٔ ${state.doc.layers.length + 1}`)
   364→      const next: PelcDocument = { ...state.doc, layers: [...state.doc.layers, layer] }
   365→      state.commit(next, "افزودن لایه")
   366→      set({ selectedLayerId: layer.id, selectedClusterStart: null, selectedLayerIds: [], activeTab: "word" })
   367→    },
   368→    deleteLayer: (id) => {
   369→      const state = get()
   370→      if (state.doc.layers.length <= 1) return
   371→      const next: PelcDocument = { ...state.doc, layers: state.doc.layers.filter((l) => l.id !== id) }
   372→      state.commit(next, "حذف لایه")
   373→      set({ selectedLayerIds: state.selectedLayerIds.filter((x) => x !== id) })
   374→    },
   375→    duplicateLayer: (id) => {
   376→      const state = get()
   377→      const src = state.doc.layers.find((l) => l.id === id)
   378→      if (!src) return
   379→      const copy: Layer = JSON.parse(JSON.stringify(src))
   380→      copy.id = `${src.id}-copy-${Date.now()}`
   381→      copy.name = `${src.name} (کپی)`
   382→      const idx = state.doc.layers.findIndex((l) => l.id === id)
   383→      const layers = [...state.doc.layers]
   384→      layers.splice(idx + 1, 0, copy)
   385→      state.commit({ ...state.doc, layers }, "تکثیر لایه")
   386→      set({ selectedLayerId: copy.id, selectedClusterStart: null, selectedLayerIds: [] })
   387→    },
   388→
   389→    splitLayerIntoWords: (id) => {
   390→      const state = get()
   391→      const layer = findLayer(state.doc, id)
   392→      if (!layer?.text) return false
   393→      const tokens = tokenizeWords(layer.text.text)
   394→      if (tokens.length < 2) return false
   395→      const boxes = measureWordPositions(layer, state.viewScale)
   396→      if (!boxes) return false
   397→
   398→      const wordLayers: Layer[] = tokens.map((tk, i) => {
   399→        const copy: Layer = JSON.parse(JSON.stringify(layer))
   400→        copy.id = newId()
   401→        copy.name = `کلمه: ${tk.token.slice(0, 16)}`
   402→        copy.text = {
   403→          ...layer.text!,
   404→          text: tk.token,
   405→          perLetter: slicePerLetter(layer.text!.perLetter, tk.rawStart, tk.token.length),
   406→        }
   407→        copy.transform = {
   408→          ...layer.transform,
   409→          tx: Math.round(layer.transform.tx + boxes[i].x),
   410→          ty: Math.round(layer.transform.ty + boxes[i].y),
   411→        }
   412→        return copy
   413→      })
   414→
   415→      const idx = state.doc.layers.findIndex((l) => l.id === id)
   416→      const layers = [...state.doc.layers]
   417→      layers.splice(idx, 1, ...wordLayers)
   418→      state.commit({ ...state.doc, layers }, "تبدیل جمله به کلماتِ مستقل")
   419→      set({ selectedLayerId: wordLayers[0].id, selectedClusterStart: null, selectedLayerIds: [] })
   420→      return true
   421→    },
   422→
   423→    nudgeSelectedLayer: (dx, dy) => {
   424→      const state = get()
   425→      // انتخاب چندگانه فعال است: همه با هم تکان می‌خورند (یک گام تاریخچه)
   426→      if (state.selectedLayerIds.length > 1) {
   427→        const ids = new Set(state.selectedLayerIds)
   428→        const next: PelcDocument = {
   429→          ...state.doc,
   430→          layers: state.doc.layers.map((l) =>
   431→            ids.has(l.id) && !l.locked
   432→              ? { ...l, transform: { ...l.transform, tx: l.transform.tx + dx, ty: l.transform.ty + dy } }
   433→              : l,
   434→          ),
   435→        }
   436→        state.commit(next, "تکان‌دادن همهٔ لایه‌ها", "nudge:multi")
   437→        return
   438→      }
   439→      const layer = findLayer(state.doc, state.selectedLayerId)
   440→      if (!layer) return
   441→      const next = withLayer(state.doc, layer.id, (l) => ({
   442→        ...l,
   443→        transform: { ...l.transform, tx: l.transform.tx + dx, ty: l.transform.ty + dy },
   444→      }))
   445→      state.commit(next, "تکان‌دادن لایه", `nudge:${layer.id}`)
   446→    },
   447→
   448→    setLayerText: (text) => {
   449→      const state = get()
   450→      const layer = findLayer(state.doc, state.selectedLayerId)
   451→      if (!layer?.text) return
   452→      const perLetter = reconcilePerLetter(text, layer.text.perLetter)
   453→      const next = withLayer(state.doc, layer.id, (l) => ({
   454→        ...l,
   455→        text: { ...l.text!, text, perLetter },
   456→      }))
   457→      state.commit(next, "ویرایش متن", `text:${layer.id}`)
   458→    },
   459→    setLayerFont: (fontId) => {
   460→      const state = get()
   461→      const layer = findLayer(state.doc, state.selectedLayerId)
   462→      if (!layer || !PELC_FONT_IDS.has(fontId)) return
   463→      const next = withLayer(state.doc, layer.id, (l) =>
   464→        l.text ? { ...l, text: { ...l.text, fontId } } : l,
   465→      )
   466→      state.commit(next, "تغییر قلم")
   467→    },
   468→    setLayerFontSize: (size) => {
   469→      const state = get()
   470→      const layer = findLayer(state.doc, state.selectedLayerId)
   471→      if (!layer) return
   472→      const next = withLayer(state.doc, layer.id, (l) =>
   473→        l.text ? { ...l, text: { ...l.text, fontSize: size } } : l,
   474→      )
   475→      state.commit(next, "اندازهٔ قلم", `size:${layer.id}`)
   476→    },
   477→    setLayerColor: (argb) => {
   478→      const state = get()
   479→      const layer = findLayer(state.doc, state.selectedLayerId)
   480→      if (!layer) return
   481→      const next = withLayer(state.doc, layer.id, (l) => ({
   482→        ...l,
   483→        style: { ...l.style, color: argb },
   484→        text: l.text ? { ...l.text, color: argb } : l.text,
   485→      }))

   487→    },

   579→    setLetterVariant: (tag) => {
   580→      const state = get()
   581→      const layer = findLayer(state.doc, state.selectedLayerId)
   582→      const start = state.selectedClusterStart
   583→      if (!layer?.text || start === null) return
   584→      const key = String(start)
   585→      const existing: LetterOverride = layer.text.perLetter[key] ?? defaultLetterOverride()
   586→      const updated: LetterOverride = { ...existing, variantTag: tag }
   587→      const perLetter = { ...layer.text.perLetter }
   588→      if (tag === null && isDefaultOverride(updated)) {
   589→        delete perLetter[key]
   590→      } else {
   591→        perLetter[key] = updated
   592→      }
   593→      const next = withLayer(state.doc, layer.id, (l) => ({ ...l, text: { ...l.text!, perLetter } }))
   594→      state.commit(next, tag ? `گونهٔ ${tag} روی حرف` : "حذف گونهٔ حرف")
   595→    },
   596→    setLetterForm: (form) => {
   597→      const state = get()
   598→      const layer = findLayer(state.doc, state.selectedLayerId)
   599→      const start = state.selectedClusterStart
   600→      if (!layer?.text || start === null) return
   601→      const key = String(start)
   602→      const existing: LetterOverride = layer.text.perLetter[key] ?? defaultLetterOverride()
   603→      const updated: LetterOverride = { ...existing, formOverride: form === "AUTO" ? null : normalizeLetterForm(form) }
   604→      const perLetter = { ...layer.text.perLetter }
   605→      if (isDefaultOverride(updated)) delete perLetter[key]
   606→      else perLetter[key] = updated
   607→      const next = withLayer(state.doc, layer.id, (l) => ({ ...l, text: { ...l.text!, perLetter } }))
   608→      state.commit(next, form === "AUTO" ? "بازگشت فرم حرف به خودکار" : `فرم حرف: ${LETTER_FORM_TITLES[form]}`)
   609→    },
   610→    setLetterFont: (fontId) => {
   611→      const state = get()
   612→      const layer = findLayer(state.doc, state.selectedLayerId)
   613→      const start = state.selectedClusterStart
   614→      if (!layer?.text || start === null) return
   615→      if (fontId !== null && !PELC_FONT_IDS.has(fontId)) return
   616→      const key = String(start)
   617→      const existing: LetterOverride = layer.text.perLetter[key] ?? defaultLetterOverride()
   618→      const updated: LetterOverride = { ...existing, fontId }
   619→      const perLetter = { ...layer.text.perLetter }
   620→      if (isDefaultOverride(updated)) delete perLetter[key]
   621→      else perLetter[key] = updated
   622→      const next = withLayer(state.doc, layer.id, (l) => ({ ...l, text: { ...l.text!, perLetter } }))
   623→      state.commit(next, fontId ? "قلمِ این حرف" : "حذف قلمِ اختصاصی حرف")
   624→    },
   625→    insertKashida: () => {
   626→      const state = get()
   627→      const layer = findLayer(state.doc, state.selectedLayerId)
   628→      const start = state.selectedClusterStart
   629→      if (!layer?.text || start === null) return
   630→      const cluster = letterClusters(layer.text.text).find((c) => c.start === start)
   631→      if (!cluster) return
   632→      if (cluster.kashidaBefore >= MAX_KASHIDA_BEFORE) return
   633→      const mutation = insertKashidaBefore(layer.text.text, start)
   634→      if (!mutation) return
   635→      commitTextMutation(mutation, "درج کشیده")
   636→    },
   637→    removeKashida: () => {
   638→      const state = get()
   639→      const layer = findLayer(state.doc, state.selectedLayerId)
   640→      const start = state.selectedClusterStart
   641→      if (!layer?.text || start === null) return
   642→      const mutation = removeKashidaBefore(layer.text.text, start)
   643→      if (!mutation) return
   644→      commitTextMutation(mutation, "حذف کشیده")
   645→    },
   646→    addMark: (type) => {
   647→      const state = get()
   648→      const layer = findLayer(state.doc, state.selectedLayerId)
   649→      const start = state.selectedClusterStart
   650→      if (!layer?.text || start === null) return
   651→      const mutation = insertMarkAfter(layer.text.text, start, markChar(type))
   652→      if (!mutation) return
   653→      // رکورد تنظیم علامت در perLetter.marks ثبت می‌شود (سازگار با مدل اندروید)
   654→      const key = String(start)
   655→      const existing = layer.text.perLetter[key] ?? defaultLetterOverride()
   656→      const perLetter0 = {
   657→        ...layer.text.perLetter,
   658→        [key]: { ...existing, marks: [...existing.marks, { type, dx: 0, dy: 0, scale: 1 }] },
   659→      }
   660→      commitTextMutation(mutation, `افزودن ${markTitle(type)}`, perLetter0)
   661→    },
   662→    removeMark: (markIndex) => {
   663→      const state = get()
   664→      const layer = findLayer(state.doc, state.selectedLayerId)
   665→      const start = state.selectedClusterStart
   666→      if (!layer?.text || start === null) return
   667→      const mutation = removeMarkAt(layer.text.text, start, markIndex)
   668→      if (!mutation) return
   669→      const key = String(start)
   670→      const existing = layer.text.perLetter[key]
   671→      const perLetter0 = { ...layer.text.perLetter }
   672→      if (existing) {
   673→        const updated = { ...existing, marks: existing.marks.filter((_, i) => i !== markIndex) }
   674→        if (isDefaultOverride(updated)) delete perLetter0[key]
   675→        else perLetter0[key] = updated
   676→      }
   677→      commitTextMutation(mutation, "حذف علامت", perLetter0)
   678→    },
   679→
   680→    clusters: () => {
   681→      const state = get()
   682→      const layer = findLayer(state.doc, state.selectedLayerId)
   683→      if (!layer?.text) return []
   684→      return letterClusters(layer.text.text)
   685→    },
   686→  }
